# Función para inicializar Conda bajo demanda
condabegin() {
    __conda_setup="$('/home/manza/miniconda3/bin/conda' 'shell.bash' 'hook' 2> /dev/null)"
    if [ $? -eq 0 ]; then
        eval "$__conda_setup"
    else
        if [ -f "/home/manza/miniconda3/etc/profile.d/conda.sh" ]; then
            . "/home/manza/miniconda3/etc/profile.d/conda.sh"
        else
            export PATH="/home/manza/miniconda3/bin:$PATH"
        fi
    fi
    unset __conda_setup
    echo "Conda inicializado correctamente."
}

extract() {
  if [ -f $1 ] ; then
    case $1 in
      *.tar.bz2)   tar xjf $1     ;;
      *.tar.gz)    tar xzf $1     ;;
      *.bz2)       bunzip2 $1     ;;
      *.rar)       unrar x $1     ;;
      *.gz)        gunzip $1      ;;
      *.tar)       tar xf $1      ;;
      *.tbz2)      tar xjf $1     ;;
      *.tgz)       tar xzf $1     ;;
      *.zip)       unzip $1       ;;
      *.Z)         uncompress $1  ;;
      *.7z)        7z x $1        ;;
      *)           echo "'$1' no puede ser extraído vía extract()" ;;
    esac
  else
    echo "'$1' no es un archivo válido"
  fi
}

# Función para cambiar el título de la ventana
setTermName() {
    if [ -z "$1" ]; then
        echo "Uso: setTermName 'nuevo nombre'"
    else
        echo -ne "\033]0;$1\007"
    fi
}
